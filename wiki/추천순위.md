# 추천 순위

문서별 추천(좋아요) 수를 집계한 순위입니다. 추천 버튼은 각 문서 상단에 있으며, 추천 수는 백엔드 API(`server/app.py`)에 저장됩니다.

<script>
(function () {
  var API = window.LIKE_API || "http://localhost:8000";
  function load() {
    var el = document.getElementById("wiki-rank");
    if (!el) return;
    el.innerHTML = "불러오는 중...";
    fetch(API + "/api/rank")
      .then(function (r) { return r.json(); })
      .then(function (list) {
        if (!list.length) {
          el.innerHTML = "아직 추천이 없습니다.";
          return;
        }
        var rows = list.map(function (item, i) {
          var name = item.path.replace(/^\//, "").replace(/\.md$/, "");
          return (
            "<tr><td>" + (i + 1) + "</td>" +
            "<td>" + name + "</td>" +
            "<td>" + item.count + "</td></tr>"
          );
        }).join("");
        el.innerHTML =
          "<table><thead><tr><th>순위</th><th>문서</th><th>추천</th></tr></thead>" +
          "<tbody>" + rows + "</tbody></table>";
      })
      .catch(function () { el.innerHTML = "순위를 불러오지 못했습니다."; });
  }
  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", load);
  } else {
    load();
  }
})();
</script>

<div id="wiki-rank"></div>
